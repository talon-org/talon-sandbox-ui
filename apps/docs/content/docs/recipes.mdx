## 1. Form with validation

Combine `FormItem` (TanStack Form wiring), `Input`, `Checkbox`, and `Button` to build a validated create-sandbox form. `FormItem` automatically surfaces the first error from `field.state.meta.errors` after the field is touched.

```tsx
import { useForm } from '@tanstack/react-form';
import { FormItem, Input, Checkbox, Button } from '@talon-sandbox/react';

function CreateSandboxForm({ onSuccess }: { onSuccess: () => void }) {
  const form = useForm({
    defaultValues: { name: '', ephemeral: false },
    onSubmit: async ({ value }) => {
      await api.createSandbox(value);
      onSuccess();
    },
  });

  return (
    <form
      onSubmit={(e) => { e.preventDefault(); form.handleSubmit(); }}
      style={{ display: 'flex', flexDirection: 'column', gap: 16 }}
    >
      <form.Field
        name="name"
        validators={{
          onChange: ({ value }) =>
            !value ? 'Name is required' :
            value.length < 3 ? 'Minimum 3 characters' :
            undefined,
        }}
      >
        {(field) => (
          <FormItem field={field} label="Sandbox name" required>
            {(f) => (
              <Input
                value={f.state.value}
                onChange={(e) => f.handleChange(e.target.value)}
                onBlur={f.handleBlur}
                placeholder="my-sandbox"
              />
            )}
          </FormItem>
        )}
      </form.Field>

      <form.Field name="ephemeral">
        {(field) => (
          <Checkbox
            checked={field.state.value}
            onChange={(checked) => field.handleChange(checked)}
            label="Ephemeral (auto-delete after session)"
          />
        )}
      </form.Field>

      <form.Subscribe selector={(s) => s.isSubmitting}>
        {(submitting) => (
          <Button type="submit" variant="primary" disabled={submitting}>
            {submitting ? 'Creating…' : 'Create sandbox'}
          </Button>
        )}
      </form.Subscribe>
    </form>
  );
}
```

## 2. Filterable data table with server pagination

Wire `DataTable` in `mode="server"` with caller-owned state. Debounce the filter to avoid per-keystroke API calls.

```tsx
import { useCallback, useState } from 'react';
import { DataTable, EmptyState, FilterBar } from '@talon-sandbox/react';
import type { ColumnDef, DataTableServerState } from '@talon-sandbox/react';
import { useDebouncedValue } from '@/hooks/useDebouncedValue';
import { useSandboxList } from '@/hooks/useSandboxList';

const COLUMNS: ColumnDef<Sandbox>[] = [
  { accessorKey: 'id', header: 'ID' },
  { accessorKey: 'name', header: 'Name' },
  { accessorKey: 'status', header: 'Status' },
  { accessorKey: 'region', header: 'Region' },
];

export function SandboxListPage() {
  const [page, setPage] = useState(0);
  const [pageSize, setPageSize] = useState(25);
  const [query, setQuery] = useState('');
  const debouncedQuery = useDebouncedValue(query, 300);

  const { data, total, isLoading } = useSandboxList({
    page, pageSize, query: debouncedQuery,
  });

  const serverState: DataTableServerState = {
    total,
    page,
    pageSize,
    onPageChange: setPage,
    onPageSizeChange: (s) => { setPageSize(s); setPage(0); },
    globalFilter: query,
    onGlobalFilterChange: (q) => { setQuery(q); setPage(0); },
  };

  return (
    <DataTable
      columns={COLUMNS}
      data={data ?? []}
      mode="server"
      serverState={serverState}
      emptyState={
        isLoading ? null : (
          <EmptyState
            title="No sandboxes found"
            description={debouncedQuery ? 'Try a different search term.' : 'Create your first sandbox.'}
          />
        )
      }
    />
  );
}
```

## 3. Confirmation flow

Open a `Dialog` to confirm a destructive action, then show a `toast` on completion.

```tsx
import { useState } from 'react';
import { Button, Dialog, toast } from '@talon-sandbox/react';

function DeleteSandboxButton({ id, name }: { id: string; name: string }) {
  const [open, setOpen] = useState(false);
  const [deleting, setDeleting] = useState(false);

  async function handleConfirm() {
    setDeleting(true);
    try {
      await api.deleteSandbox(id);
      setOpen(false);
      toast.success(`"${name}" deleted`);
    } catch (err) {
      toast.error('Delete failed — please try again');
    } finally {
      setDeleting(false);
    }
  }

  return (
    <>
      <Button variant="danger" onClick={() => setOpen(true)}>
        Delete
      </Button>
      <Dialog
        open={open}
        onClose={() => setOpen(false)}
        title={`Delete "${name}"?`}
        description="This action cannot be undone. All data in this sandbox will be permanently removed."
        footer={
          <>
            <Button variant="default" onClick={() => setOpen(false)}>
              Cancel
            </Button>
            <Button variant="danger" onClick={handleConfirm} disabled={deleting}>
              {deleting ? 'Deleting…' : 'Delete'}
            </Button>
          </>
        }
      />
    </>
  );
}
```

## 4. Detail drawer

Slide in a `Drawer` with sandbox details arranged via `Tabs` and `KV` rows.

```tsx
import { Drawer, Tabs, KV, Badge } from '@talon-sandbox/react';

function SandboxDetailDrawer({
  sandbox,
  open,
  onClose,
}: {
  sandbox: Sandbox;
  open: boolean;
  onClose: () => void;
}) {
  return (
    <Drawer
      open={open}
      onClose={onClose}
      title={sandbox.name}
      width={480}
    >
      <Tabs
        items={[
          {
            key: 'overview',
            label: 'Overview',
            children: (
              <div style={{ display: 'flex', flexDirection: 'column', gap: 4 }}>
                <KV label="ID" value={sandbox.id} copy />
                <KV label="Status" value={<Badge variant="success">{sandbox.status}</Badge>} />
                <KV label="Region" value={sandbox.region} />
                <KV label="Image" value={sandbox.image} />
                <KV label="Created" value={new Date(sandbox.createdAt).toLocaleString()} />
              </div>
            ),
          },
          {
            key: 'resources',
            label: 'Resources',
            children: (
              <div style={{ display: 'flex', flexDirection: 'column', gap: 4 }}>
                <KV label="CPU" value={`${sandbox.cpu} vCPU`} />
                <KV label="Memory" value={`${sandbox.memory} GiB`} />
                <KV label="Disk" value={`${sandbox.disk} GiB`} />
              </div>
            ),
          },
        ]}
      />
    </Drawer>
  );
}
```

## 5. Empty / loading / error state machine

Drive `EmptyState` from a single status value so the UI transitions cleanly between loading, empty, and error states.

```tsx
import { EmptyState, ProgressBar } from '@talon-sandbox/react';

type ListStatus = 'loading' | 'empty' | 'error' | 'populated';

function deriveStatus(
  isLoading: boolean,
  isError: boolean,
  count: number,
): ListStatus {
  if (isLoading) return 'loading';
  if (isError) return 'error';
  if (count === 0) return 'empty';
  return 'populated';
}

function SandboxListShell({
  isLoading,
  isError,
  retry,
  children,
}: {
  isLoading: boolean;
  isError: boolean;
  retry: () => void;
  children: React.ReactNode;
}) {
  const count = React.Children.count(children);
  const status = deriveStatus(isLoading, isError, count);

  if (status === 'loading') {
    return (
      <div aria-busy="true" aria-label="Loading sandboxes">
        <ProgressBar indeterminate aria-label="Loading" />
      </div>
    );
  }

  if (status === 'error') {
    return (
      <EmptyState
        title="Failed to load sandboxes"
        description="Check your connection and try again."
        action={{ label: 'Retry', onClick: retry }}
      />
    );
  }

  if (status === 'empty') {
    return (
      <EmptyState
        title="No sandboxes yet"
        description="Create your first sandbox to get started."
        action={{ label: 'New sandbox', href: '/sandboxes/new' }}
      />
    );
  }

  return <>{children}</>;
}
```
